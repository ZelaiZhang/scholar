# Research OS Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local-first, evidence-traceable Codex research workspace that guides a student from topic definition through literature synthesis, study design, result interpretation, writing, review, and rebuttal without running experiments.

**Architecture:** Keep durable research state in Markdown, YAML, JSON, and CSV. Use a small Python package for deterministic project creation, source registration, PDF extraction, OpenAI-compatible model calls, and evidence validation; use focused Codex skills for judgment-heavy research workflows.

**Tech Stack:** Python 3.11+, `argparse`, `dataclasses`, `urllib`, PyYAML, pypdf, pytest, Markdown/YAML/CSV, Git.

```python
    "research-brief.md": "00-research-brief.md",

def create_project(root: Path, title: str, slug: str, template_root: Path | None = None) -> Path:
    validate_slug(slug)
    destination = root / "projects" / slug
    if destination.exists():
        raise ProjectExistsError(f"课题已存在: {destination}")
    destination.mkdir(parents=True)
    for source_name, target_name in PROJECT_FILES.items():
        content = (template_root / source_name).read_text(encoding="utf-8")
        atomic_write_text(destination / target_name, content.replace("{{PROJECT_TITLE}}", title))
    for name in ("writing", "reviews", "artifacts"):
        folder = destination / name
        folder.mkdir()
        (folder / ".gitkeep").touch()
    return destination
```

```python
@dataclass(frozen=True)
class ExtractedPDF:
    path: Path
    pages: tuple[str, ...]

    @property
    def markdown(self) -> str:
        return "\n\n".join(
            f"<!-- page:{index} -->\n{text}" for index, text in enumerate(self.pages, 1)
        )

def extract_pdf(path: Path, reader_factory=None) -> ExtractedPDF:
    if not path.is_file():
        raise FileNotFoundError(path)
    if reader_factory is None:
        from pypdf import PdfReader
        reader_factory = PdfReader
    pages = tuple((page.extract_text() or "").strip() for page in reader_factory(str(path)).pages)
    if not any(pages):
        raise NeedsOCRError("PDF 没有可提取文本，请先进行 OCR")
    return ExtractedPDF(path=path, pages=pages)
```

Add `extract-pdf <path> --output <path>` to the CLI. Refuse to overwrite an existing output unless `--force` is present.

```python
@dataclass(frozen=True)
class CompletionResult:
    content: str
    provenance: dict[str, object]

class OpenAICompatibleProvider:
    def complete(self, system: str, user: str, *, external_api_allowed: bool) -> CompletionResult:
        if not external_api_allowed:
            raise PermissionError("该来源未授权发送到外部 API")
        api_key = os.getenv(self.api_key_env)
        if not api_key:
            raise MissingAPIKey(f"请设置环境变量 {self.api_key_env}")
        payload = {
            "model": self.model,
            "messages": [{"role": "system", "content": system}, {"role": "user", "content": user}],
            "temperature": self.temperature,
        }
        raw = self.transport(f"{self.base_url}/chat/completions", {"Authorization": f"Bearer {api_key}"}, payload, self.timeout)
        return CompletionResult(raw["choices"][0]["message"]["content"], {
            "provider_base_url": self.base_url,
            "model": self.model,
            "created_at": datetime.now(timezone.utc).isoformat(),
            "temperature": self.temperature,
            "usage": raw.get("usage", {}),
        })
```

Implement the default HTTP transport with `urllib.request`, JSON encoding, timeout handling, and sanitized errors. Add `model-call` CLI support that reads prompts from files and writes content plus a sibling provenance JSON atomically.

```python
ALLOWED_TYPES = {"fact", "inference", "hypothesis"}
ALLOWED_STATUS = {"unverified", "partially_verified", "verified", "conflicted"}

@dataclass(frozen=True)
class ValidationIssue:
    code: str
    claim_id: str
    message: str

def validate_ledger(ledger: dict[str, object]) -> list[ValidationIssue]:
    issues: list[ValidationIssue] = []
    for claim in ledger.get("claims", []):
        claim_id = str(claim.get("claim_id", ""))
        if claim.get("type") not in ALLOWED_TYPES:
            issues.append(ValidationIssue("invalid_type", claim_id, "claim 类型必须显式标记"))
        if claim.get("status") not in ALLOWED_STATUS:
            issues.append(ValidationIssue("invalid_status", claim_id, "未知核验状态"))
        if claim.get("type") == "fact" and claim.get("status") == "verified":
            for source in claim.get("support", []):
                if not str(source.get("locator", "")).strip():
                    issues.append(ValidationIssue("missing_locator", claim_id, "已核验事实缺少页码或章节"))
        if not str(claim.get("limitations", "")).strip():
            issues.append(ValidationIssue("missing_limitations", claim_id, "必须陈述证据限制"))
    return issues
```

Add duplicate claim-ID detection, missing support detection for verified facts, invalid confidence detection, YAML loading, and `validate-ledger <path> --report <path>` to the CLI. Exit with code 1 on issues and 0 on success.
